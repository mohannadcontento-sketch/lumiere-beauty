   160→    const totalEarned = ledger
   161→      .filter((l) => l.points > 0)

   163→    const totalRedeemed = Math.abs(
   164→      ledger.filter((l) => l.points < 0).reduce((s, l) => s + l.points, 0),

   169→    return { totalEarned, totalRedeemed, activeCount, avg };

   173→    () => [...customers].sort((a, b) => b.loyaltyPoints - a.loyaltyPoints).slice(0, 5),

   179→      {/* ---- Stat cards ---- */}
   180→      <div className="grid grid-cols-2 gap-4 lg:grid-cols-4">

   494→type LedgerTypeFilter = "all" | "earn" | "redeem";
   495→
   496→function LedgerTab({
   497→  ledger,
   498→  customers,
   499→  isReadOnly,
   500→}: {
   501→  ledger: LoyaltyLedgerEntry[];
   502→  customers: Customer[];
   503→  isReadOnly: boolean;
   504→}) {
   505→  const [customerFilter, setCustomerFilter] = useState<string>("all");
   506→  const [typeFilter, setTypeFilter] = useState<LedgerTypeFilter>("all");
   507→
   508→  const customerName = (id: string) =>
   509→    customers.find((c) => c.id === id)?.name ?? "—";
   510→
   511→  const filtered = useMemo(() => {
   512→    return ledger.filter((l) => {
   513→      if (customerFilter !== "all" && l.customerId !== customerFilter) return false;
   514→      if (typeFilter === "earn" && l.type !== "earn") return false;
   515→      if (typeFilter === "redeem" && l.type !== "redeem" && l.type !== "adjust") return false;
   516→      return true;
   517→    });
   518→  }, [ledger, customerFilter, typeFilter]);
   519→
   520→  return (
   521→    <div className="space-y-4">
   522→      {/* ---- Filter bar ---- */}
   523→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center">
   524→        <div className="flex flex-1 items-center gap-2">
   525→          <Filter className="size-4 text-muted-foreground" />
   526→          <Select value={customerFilter} onValueChange={setCustomerFilter}>
   527→            <SelectTrigger size="sm" className="w-full sm:w-60">
   528→              <SelectValue placeholder="كل العملاء" />
   529→            </SelectTrigger>
   530→            <SelectContent>
   531→              <SelectItem value="all">كل العملاء</SelectItem>
   532→              {customers.map((c) => (
   533→                <SelectItem key={c.id} value={c.id}>
   534→                  {c.name}
   535→                </SelectItem>
   536→              ))}
   537→            </SelectContent>
   538→          </Select>
   539→        </div>
   540→        <div className="flex items-center gap-1 rounded-md border border-border bg-card p-1">
   541→          {(["all", "earn", "redeem"] as LedgerTypeFilter[]).map((t) => (
   542→            <button
   543→              key={t}
   544→              type="button"
   545→              onClick={() => setTypeFilter(t)}
   546→              className={`rounded px-3 py-1 text-xs font-medium transition-colors ${
   547→                typeFilter === t
   548→                  ? "bg-primary text-primary-foreground"
   549→                  : "text-muted-foreground hover:text-foreground"
   550→              }`}
   551→            >
   552→              {t === "all" ? "الكل" : t === "earn" ? "مكتسب" : "مستبدل"}
   553→            </button>
   554→          ))}
   555→        </div>
   556→        <span className="hidden text-xs text-muted-foreground sm:inline">
   557→          {formatNumber(filtered.length)} سجل
   558→        </span>
   559→      </div>
   560→
   561→      {/* ---- Table ---- */}
   562→      {filtered.length === 0 ? (
   563→        <EmptyState
   564→          icon={<Coins className="size-10" />}
   565→          title="لا سجلات مطابقة"
   566→          description="لا توجد حركات نقاط مطابقة لعوامل التصفية الحالية."
   567→        />
   568→      ) : (
   569→        <div className="card-hover rounded-lg border border-border bg-card">
   570→          <div className="max-h-[600px] overflow-y-auto thin-scroll">
   571→            <table className="w-full min-w-[720px] divide-y divide-border text-sm">
   572→              <thead className="sticky top-0 z-10 bg-card">
   573→                <tr className="text-right">
   574→                  <Th>التاريخ</Th>
   575→                  <Th>العميل</Th>
   576→                  <Th className="text-center">النوع</Th>
   577→                  <Th className="text-center">النقاط</Th>
   578→                  <Th>السبب</Th>
   579→                </tr>
   580→              </thead>
   581→              <tbody className="divide-y divide-border">
   582→                {filtered.map((l) => {
   583→                  const isEarn = l.type === "earn";
   584→                  const isRedeem = l.type === "redeem";
   585→                  const isAdjust = l.type === "adjust";
   586→                  return (
   587→                    <tr key={l.id} className="transition-colors hover:bg-muted/40">
   588→                      {/* Date */}
   589→                      <td className="whitespace-nowrap px-4 py-3 text-xs text-muted-foreground">
   590→                        {formatDateTime(l.date)}
   591→                      </td>
   592→                      {/* Customer */}
   593→                      <td className="px-4 py-3 font-medium text-foreground">
   594→                        {customerName(l.customerId)}
   595→                      </td>
   596→                      {/* Type */}
   597→                      <td className="px-4 py-3 text-center">
   598→                        {isEarn ? (
   599→                          <Badge variant="success">مكتسب</Badge>
   600→                        ) : isRedeem ? (
   601→                          <Badge variant="warning">مستبدل</Badge>
   602→                        ) : (
   603→                          <Badge variant="neutral">تعديل</Badge>
   604→                        )}
   605→                      </td>
   606→                      {/* Points */}
   607→                      <td className="px-4 py-3 text-center">
   608→                        <span
   609→                          className={`tnum font-display text-base font-bold ${
   610→                            isEarn || (isAdjust && l.points > 0)
   611→                              ? "text-emerald-700"
   612→                              : "text-amber-700"
   613→                          }`}
   614→                        >
   615→                          {l.points > 0 ? "+" : ""}
   616→                          {formatNumber(l.points)}
   617→                        </span>
   618→                      </td>
   619→                      {/* Reason */}
   620→                      <td className="px-4 py-3 text-xs text-muted-foreground">
   621→                        {l.reason}
   622→                      </td>
   623→                    </tr>
   624→                  );
   625→                })}
   626→              </tbody>
   627→            </table>
   628→          </div>
   629→        </div>
   630→      )}
   631→
   632→      {/* ---- Customer-context note ---- */}
   633→      {isReadOnly && (
   634→        <p className="text-center text-xs text-muted-foreground">
   635→          هذه قائمة بحركات نقاطك فقط. للاستبدال، تواصلي مع الاستقبال.
   636→        </p>
   637→      )}
   638→    </div>
   639→  );
   640→}

   658→// ============================================================
   659→// Redeem dialog (owner / reception only)
   660→// ============================================================
   661→function RedeemDialogButton({
   662→  customers,
   663→  rule,
   664→}: {
   665→  customers: Customer[];
   666→  rule: LoyaltyRule;
   667→}) {
   668→  const redeemPoints = useApp((s) => s.redeemPoints);
   669→  const [open, setOpen] = useState(false);
   670→  const [customerId, setCustomerId] = useState<string>("");
   671→  const [points, setPoints] = useState<string>("");
   672→  const [reason, setReason] = useState<string>("");
   673→
   674→  const selectedCustomer = customers.find((c) => c.id === customerId);
   675→  const pointsToRedeem = Number(points) || 0;
   676→  const canRedeem =
   677→    !!customerId &&
   678→    pointsToRedeem >= rule.minRedeemPoints &&
   679→    selectedCustomer &&
   680→    pointsToRedeem <= selectedCustomer.loyaltyPoints &&
   681→    reason.trim().length > 0;
   682→
   683→  const eligibleCustomers = useMemo(
   684→    () => customers.filter((c) => c.loyaltyPoints >= rule.minRedeemPoints),
   685→    [customers, rule.minRedeemPoints],
   686→  );
   687→
   688→  const reset = () => {
   689→    setCustomerId("");
   690→    setPoints("");
   691→    setReason("");
   692→  };
   693→
   694→  const handleConfirm = () => {
   695→    if (!canRedeem || !selectedCustomer) return;
   696→    const res = redeemPoints(selectedCustomer.id, pointsToRedeem, reason.trim());
   697→    if (res.ok) {
   698→      toast.success(
   699→        `تم استبدال ${formatNumber(pointsToRedeem)} نقطة لـ ${selectedCustomer.name}`,
   700→      );
   701→      setOpen(false);
   702→      reset();
   703→    } else {
   704→      toast.error(res.error ?? "تعذّر الاستبدال");
   705→    }
   706→  };
   707→
   708→  return (
   709→    <>
   710→      <Button size="sm" onClick={() => setOpen(true)}>
   711→        <Gift className="size-4" />
   712→        استبدال نقاط
   713→      </Button>
   714→
   715→      <Dialog
   716→        open={open}
   717→        onOpenChange={(o) => {
   718→          setOpen(o);
   719→          if (!o) reset();
   720→        }}
   721→      >
   722→        <DialogContent className="sm:max-w-md">
   723→          <DialogHeader>
   724→            <DialogTitle>استبدال نقاط الولاء</DialogTitle>
   725→            <DialogDescription>
   726→              استبدل نقاط عميل بمبلغ خصم. الحد الأدنى {formatNumber(rule.minRedeemPoints)} نقطة، وقيمة النقطة {formatEGP(rule.pointsToEgpRate)}.
   727→            </DialogDescription>
   728→          </DialogHeader>
   729→
   730→          <div className="space-y-4">
   731→            {/* Customer */}
   732→            <div className="space-y-1.5">
   733→              <Label htmlFor="redeem-customer">العميل</Label>
   734→              <Select value={customerId} onValueChange={setCustomerId}>
   735→                <SelectTrigger id="redeem-customer" className="w-full">
   736→                  <SelectValue placeholder="اختاري العميل" />
   737→                </SelectTrigger>
   738→                <SelectContent>
   739→                  {eligibleCustomers.length === 0 ? (
   740→                    <SelectItem value="__none" disabled>
   741→                      لا يوجد عملاء مؤهلون
   742→                    </SelectItem>
   743→                  ) : (
   744→                    eligibleCustomers.map((c) => (
   745→                      <SelectItem key={c.id} value={c.id}>
   746→                        {c.name} — {formatNumber(c.loyaltyPoints)} نقطة
   747→                      </SelectItem>
   748→                    ))
   749→                  )}
   750→                </SelectContent>
   751→              </Select>
   752→              {selectedCustomer && (
   753→                <p className="text-xs text-muted-foreground">
   754→                  الرصيد الحالي:{" "}
   755→                  <span className="font-semibold text-foreground tnum">
   756→                    {formatNumber(selectedCustomer.loyaltyPoints)}
   757→                  </span>{" "}
   758→                  نقطة
   759→                </p>
   760→              )}
   761→            </div>
   762→
   763→            {/* Points */}
   764→            <div className="space-y-1.5">
   765→              <Label htmlFor="redeem-points">عدد النقاط للاستبدال</Label>
   766→              <Input
   767→                id="redeem-points"
   768→                type="number"
   769→                inputMode="numeric"
   770→                min={rule.minRedeemPoints}
   771→                max={selectedCustomer?.loyaltyPoints ?? 0}
   772→                value={points}
   773→                onChange={(e) => setPoints(e.target.value)}
   774→                placeholder={`الحد الأدنى ${rule.minRedeemPoints}`}
   775→                dir="ltr"
   776→                className="text-end"
   777→              />
   778→              {pointsToRedeem > 0 && (
   779→                <p className="text-xs text-muted-foreground">
   780→                  قيمة الخصم:{" "}
   781→                  <span className="font-semibold text-foreground tnum">
   782→                    {formatEGP(pointsToRedeem * rule.pointsToEgpRate)}
   783→                  </span>
   784→                </p>
   785→              )}
   786→            </div>
   787→
   788→            {/* Reason */}
   789→            <div className="space-y-1.5">
   790→              <Label htmlFor="redeem-reason">السبب</Label>
   791→              <Input
   792→                id="redeem-reason"
   793→                value={reason}
   794→                onChange={(e) => setReason(e.target.value)}
   795→                placeholder="مثال: استبدال مقابل خصم فاتورة"
   796→              />
   797→            </div>
   798→          </div>
   799→
   800→          <DialogFooter>
   801→            <Button variant="outline" onClick={() => setOpen(false)}>
   802→              إلغاء
   803→            </Button>
   804→            <Button onClick={handleConfirm} disabled={!canRedeem}>
   805→              تأكيد الاستبدال
   806→            </Button>
   807→          </DialogFooter>
   808→        </DialogContent>
   809→      </Dialog>
   810→    </>
   811→  );
   812→}
