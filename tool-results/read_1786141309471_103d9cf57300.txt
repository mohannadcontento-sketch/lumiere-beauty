   442→  const taxPct = 14;

   444→  const taxAmount = Math.round((taxable * taxPct) / 100);
   445→  const total = taxable + taxAmount;
   446→  const pointsEarned = Math.floor(total / 100);
   447→  const method = paymentMethods[i % paymentMethods.length];

   479→    id: `t${i + 1}`,
   480→    code: `INV-${txnCounter++}`,
   481→    type: "sale",
   482→    customerId: cust.id,

   486→    subtotal,
   487→    discountPct,
   488→    discountAmount,
   489→    taxPct,
   490→    taxAmount,
   491→    total,
   492→    paid: isPaid ? total : 0,
   493→    paymentStatus: isPaid ? "paid" : i % 11 === 0 ? "partial" : "unpaid",
